# Hand-computed expectations for the TMS320C2x/C5x p-code, worked out from the instruction
# descriptions in SPRU014C (C2x) and SPRU056D (C5x), independent of the SLEIGH. Encodings were
# cross-checked with the Macroassembler AS. Format: see TmsEmuTest.java. Code starts at prog:0100.

# ---------------------------------------------------------------- TMS320C2x
case c2x_add_shift c2x            # LDPK 0; LAC 60h,4; ADD 61h,8  (SXM=1 sign-extends 8001h)
code C800 2460 0861
set data:0060 0x8001
set data:0061 0x0123
expect ACC 0xfff92310
expect C 0
expect OV 0
end
case c2x_sxm_off c2x              # RSXM; LDPK 0; LAC 60h,4  (no sign extension)
code CE06 C800 2460
set data:0060 0x8001
expect ACC 0x00080010
end
case c2x_addh_ovf c2x             # ZALH/ADDH overflow; ADDH never clears C
code C800 4060 4861

set data:0061 0x0001
expect ACC 0x80000000
expect OV 1
expect C 1
end
case c2x_addh_sat c2x             # SOVM: overflow saturates to 7FFFFFFFh

expect ACC 0x7fffffff
expect OV 1
expect OVM 1
end
